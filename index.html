      localStorage.setItem('alphaPortfolio', JSON.stringify(portfolio));

    function togglePrivacyMode() {

      isPrivacyMode = !isPrivacyMode;

      const btnText = document.getElementById('privacyText');

      const btnIcon = document.getElementById('privacyIcon');

      if (isPrivacyMode) {

        btnText.innerText = "Privaten Modus beenden (Zahlen anzeigen)";

    function resetToDefaults() {

      if(confirm("Möchten Sie das Portfolio wirklich auf den Ausgangszustand zurücksetzen?")) {

        portfolio = JSON.parse(JSON.stringify(DEFAULT_PORTFOLIO));

        saveToStorage();

        render();

      }

    }


    function updateItem(index, key, value) {

      portfolio[index][key] = parseFloat(value) || 0;

      saveToStorage();

      render();

    }


    function deleteItem(index) {

      portfolio.splice(index, 1);

      saveToStorage();

      render();

    }


    function handleAddPosition(e) {

      e.preventDefault();

      const ticker = document.getElementById('newTicker').value.toUpperCase().trim();

      const cluster = document.getElementById('newCluster').value;

      const shares = parseFloat(document.getElementById('newShares').value);

      const buyPrice = parseFloat(document.getElementById('newBuyPrice').value);

      const currentPrice = parseFloat(document.getElementById('newPrice').value);


      portfolio.push({ ticker, cluster, shares, buyPrice, currentPrice });

      saveToStorage();

      e.target.reset();

      render();

    }


    function render() {

      let totalCost = 0;

      let totalValue = 0;


      portfolio.forEach(item => {

        totalCost += (item.shares * item.buyPrice);

        totalValue += (item.shares * item.currentPrice);

      });


      const totalGain = totalValue - totalCost;

      const totalGainPct = totalCost > 0 ? (totalGain / totalCost) * 100 : 0;


      // 2. KPIs rendern

      document.getElementById('kpiCount').innerText = portfolio.length;


      if (isPrivacyMode) {

        document.getElementById('kpiUnrealized').className = `text-2xl font-bold mt-1 ${totalGainPct >= 0 ? 'text-emerald-400' : 'text-rose-400'}`;

    function renderCharts(totalValue) {

      // Cluster Aggregation

      const clusterMap = {};

      portfolio.forEach(item => {

        const val = item.shares * item.currentPrice;

        clusterMap[item.cluster] = (clusterMap[item.cluster] || 0) + val;

      });


      const clusterLabels = Object.keys(clusterMap);

      const clusterData = clusterLabels.map(label => ((clusterMap[label] / totalValue) * 100).toFixed(2));


      // Top Holdings

      const topItems = [...portfolio].slice(0, 7);

      const otherValue = portfolio.slice(7).reduce((acc, curr) => acc + (curr.shares * curr.currentPrice), 0);

      

      const assetLabels = topItems.map(i => i.ticker);

      const assetData = topItems.map(i => ((i.shares * i.currentPrice / totalValue) * 100).toFixed(2));

      if (otherValue > 0) {

        assetLabels.push("Weitere");

        assetData.push(((otherValue / totalValue) * 100).toFixed(2));

      }


      const colors = ['#06b6d4', '#3b82f6', '#8b5cf6', '#ec4899', '#f59e0b', '#10b981', '#64748b', '#475569'];


      // Sector Chart

      if (sectorChartInstance) sectorChartInstance.destroy();

      const ctxSector = document.getElementById('sectorChart').getContext('2d');

      sectorChartInstance = new Chart(ctxSector, {

        type: 'doughnut',

        data: {

          labels: clusterLabels,

          datasets: [{

            data: clusterData,

            backgroundColor: colors,

            borderWidth: 0

          }]

        },

        options: {

          responsive: true,

          maintainAspectRatio: false,

          plugins: {

            legend: { position: 'right', labels: { color: '#cbd5e1', font: { size: 11 } } },

            tooltip: {

              callbacks: {

                label: (c) => ` ${c.label}: ${c.raw}%`

              }

            }

          }

        }

      });


      // Assets Chart

      if (assetsChartInstance) assetsChartInstance.destroy();

      assetsChartInstance = new Chart(ctxAsset, {

        type: 'doughnut',

        data: {

          datasets: [{

            data: assetData,

            backgroundColor: colors,

            borderWidth: 0

          }]

        },

        options: {

          responsive: true,

          maintainAspectRatio: false,

          plugins: {

              }

            }

          }

        }

      });

    }
